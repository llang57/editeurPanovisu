86 public void verifyOpenRouterModels(String apiKey) {
87 if (openRouterConfig == null || !openRouterConfig.isVerifyAtStartup()) {
88 return;
89 }
90
91 System.out.println("[Config] 🔍 Vérification modèles OpenRouter...");
92
93 try {
94 // Appeler l'API pour lister les modèles disponibles
95 URI uri = URI.create("https://openrouter.ai/api/v1/models");
96 HttpURLConnection conn = (HttpURLConnection) uri.toURL().openConnection();
97 conn.setRequestMethod("GET");
98 conn.setRequestProperty("Authorization", "Bearer " + apiKey);
99 conn.setConnectTimeout(5000);
100 conn.setReadTimeout(10000);
101
102 int responseCode = conn.getResponseCode();
103 if (responseCode == 200) {
104 String response = new String(conn.getInputStream().readAllBytes(), StandardCharsets.UTF_8);
105 JsonObject jsonResponse = JsonParser.parseString(response).getAsJsonObject();
106 JsonArray dataArray = jsonResponse.getAsJsonArray("data");
107
108 Set<String> availableIds = new HashSet<>();
109 for (int i = 0; i < dataArray.size(); i++) {
110 JsonObject model = dataArray.get(i).getAsJsonObject();
111 availableIds.add(model.get("id").getAsString());
112 }
113
114 // Marquer les modèles disponibles
115 int availableCount = 0;
116 for (ModelConfig.ModelEntry entry : openRouterConfig.getModels()) {
117 boolean available = availableIds.contains(entry.getId());
118 entry.setAvailable(available);
119 if (available && entry.isEnabled()) {
120 availableCount++;
121 } else if (!available && entry.isEnabled()) {
122 System.out.println("[Config] ⚠️ Modèle indisponible: " + entry.getId());
123 }
124 }
125
126 System.out.println("[Config] ✓ OpenRouter: " + availableCount + "/" +
127 openRouterConfig.getEnabledModels().size() + " modèles disponibles");
128
129 verificationCache.put("openrouter", Instant.now());
130
131 } else {
132 System.err.println("[Config] ✗ Erreur API OpenRouter: " + responseCode);
133 }
134
135 conn.disconnect();
136
137 } catch (Exception e) {
138 System.err.println("[Config] ✗ Erreur vérification OpenRouter: " + e.getMessage());

145 public void verifyOllamaModels() {
146 if (ollamaConfig == null || !ollamaConfig.isVerifyAtStartup()) {
147 return;
148 }
149
150 System.out.println("[Config] 🔍 Vérification modèles Ollama...");
151
152 try {
153 String serverUrl = ollamaConfig.getServerUrl();
154 if (serverUrl == null) serverUrl = "http://localhost:11434";
155
156 URI uri = URI.create(serverUrl + "/api/tags");
157 HttpURLConnection conn = (HttpURLConnection) uri.toURL().openConnection();
158 conn.setRequestMethod("GET");
159 conn.setConnectTimeout(2000);
160 conn.setReadTimeout(5000);
161
162 int responseCode = conn.getResponseCode();
163 if (responseCode == 200) {
164 String response = new String(conn.getInputStream().readAllBytes(), StandardCharsets.UTF_8);
165 JsonObject jsonResponse = JsonParser.parseString(response).getAsJsonObject();
166 JsonArray models = jsonResponse.getAsJsonArray("models");
167
168 Set<String> availableNames = new HashSet<>();
169 for (int i = 0; i < models.size(); i++) {
170 JsonObject model = models.get(i).getAsJsonObject();
171 String name = model.get("name").getAsString();
172 availableNames.add(name);
173 // Ajouter aussi sans :latest pour matching flexible
174 if (name.endsWith(":latest")) {
175 availableNames.add(name.replace(":latest", ""));
176 } else {
177 availableNames.add(name + ":latest");
178 }
179 }
180
181 // Marquer les modèles disponibles
182 int availableCount = 0;
183 for (ModelConfig.ModelEntry entry : ollamaConfig.getModels()) {
184 String id = entry.getId();
185 String idWithoutTag = id.replace(":latest", "");
186 boolean available = availableNames.contains(id) ||
187 availableNames.contains(idWithoutTag);
188 entry.setAvailable(available);
189 if (available && entry.isEnabled()) {
190 availableCount++;
191 } else if (!available && entry.isEnabled()) {
192 System.out.println("[Config] ⚠️ Modèle non installé: " + entry.getId());
193 }
194 }
195
196 System.out.println("[Config] ✓ Ollama: " + availableCount + "/" +
197 ollamaConfig.getEnabledModels().size() + " modèles disponibles");
198
199 verificationCache.put("ollama", Instant.now());
200
201 } else {
202 System.err.println("[Config] ✗ Serveur Ollama inaccessible: " + responseCode);
203 }
204
205 conn.disconnect();

207 } catch (Exception e) {
208 System.err.println("[Config] ⚠️ Ollama non accessible: " + e.getMessage());
