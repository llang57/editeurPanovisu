64 private ModelConfig chargeCatalogue(String chemin, String libelle, Supplier<ModelConfig> repli) {
65 try {
66 String json = Files.readString(Path.of(chemin), StandardCharsets.UTF_8);
67 ModelConfig config = gson.fromJson(json, ModelConfig.class);
68 if (config != null && config.getModels() != null && !config.getModels().isEmpty()) {
69 System.out.println("[Config] ✓ Catalogue " + libelle + " chargé depuis " + chemin
70 + " : " + config.getModels().size() + " modèles");
71 return config;
72 }
73 System.err.println("[Config] ⚠ Catalogue " + libelle + " présent mais vide : "

162 public void verifyOpenRouterModels(String apiKey) {
163 if (openRouterConfig == null || !openRouterConfig.isVerifyAtStartup()) {
164 return;
165 }
166
167 System.out.println("[Config] 🔍 Vérification modèles OpenRouter...");
168
169 try {
170 // Appeler l'API pour lister les modèles disponibles
171 URI uri = URI.create("https://openrouter.ai/api/v1/models");
172 HttpURLConnection conn = (HttpURLConnection) uri.toURL().openConnection();
173 conn.setRequestMethod("GET");
174 conn.setRequestProperty("Authorization", "Bearer " + apiKey);
175 conn.setConnectTimeout(5000);
176 conn.setReadTimeout(10000);
177
178 int responseCode = conn.getResponseCode();
179 if (responseCode == 200) {
180 String response = new String(conn.getInputStream().readAllBytes(), StandardCharsets.UTF_8);
181 JsonObject jsonResponse = JsonParser.parseString(response).getAsJsonObject();
182 JsonArray dataArray = jsonResponse.getAsJsonArray("data");
183
184 Set<String> availableIds = new HashSet<>();
185 for (int i = 0; i < dataArray.size(); i++) {
186 JsonObject model = dataArray.get(i).getAsJsonObject();
187 availableIds.add(model.get("id").getAsString());
188 }
189
190 // Marquer les modèles disponibles
191 int availableCount = 0;
192 for (ModelConfig.ModelEntry entry : openRouterConfig.getModels()) {
193 boolean available = availableIds.contains(entry.getId());
194 entry.setAvailable(available);
195 if (available && entry.isEnabled()) {
196 availableCount++;
197 } else if (!available && entry.isEnabled()) {
198 System.out.println("[Config] ⚠️ Modèle indisponible: " + entry.getId());
199 }
200 }
201
202 System.out.println("[Config] ✓ OpenRouter: " + availableCount + "/" +
203 openRouterConfig.getEnabledModels().size() + " modèles disponibles");
204
205 verificationCache.put("openrouter", Instant.now());
206
207 } else {
208 System.err.println("[Config] ✗ Erreur API OpenRouter: " + responseCode);
209 }
210
211 conn.disconnect();
212
213 } catch (Exception e) {
214 System.err.println("[Config] ✗ Erreur vérification OpenRouter: " + e.getMessage());
215 }

239 if (responseCode == 200) {

241 JsonObject jsonResponse = JsonParser.parseString(response).getAsJsonObject();
242 JsonArray models = jsonResponse.getAsJsonArray("models");
243
244 Set<String> availableNames = new HashSet<>();
245 for (int i = 0; i < models.size(); i++) {
246 JsonObject model = models.get(i).getAsJsonObject();
247 String name = model.get("name").getAsString();
248 availableNames.add(name);
249 // Ajouter aussi sans :latest pour matching flexible
250 if (name.endsWith(":latest")) {
251 availableNames.add(name.replace(":latest", ""));
252 } else {
253 availableNames.add(name + ":latest");
254 }
255 }
256
257 // Marquer les modèles disponibles
258 int availableCount = 0;
259 for (ModelConfig.ModelEntry entry : ollamaConfig.getModels()) {
260 String id = entry.getId();
261 String idWithoutTag = id.replace(":latest", "");
262 boolean available = availableNames.contains(id) ||

264 entry.setAvailable(available);
