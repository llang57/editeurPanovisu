53 private MeshView createQuadFace(float x1, float y1, float z1,

56 float x4, float y4, float z4) {

60 mesh.getPoints().addAll(

65 );
66

68 mesh.getTexCoords().addAll(

71 1f, 1f, // 2 : bottom-right
72 0f, 1f // 3 : bottom-left
73 );
74
75 // Deux triangles formant le quad
76 mesh.getFaces().addAll(
77 0,0, 1,1, 2,2, // triangle 1
78 0,0, 2,2, 3,3 // triangle 2
79 );

214 private void applyCubeFaces(Image[] cubeFaces) {
215 // equi2cubeAuto : [0]=Front [1]=Behind [2]=Right [3]=Left [4]=Top [5]=Bottom
216 // Nos indices : FRONT=0 BACK=1 LEFT=2 RIGHT=3 TOP=4 BOTTOM=5
217 materials[FRONT] .setSelfIlluminationMap(cubeFaces[0]);
218 materials[BACK] .setSelfIlluminationMap(cubeFaces[1]);
219 materials[LEFT] .setSelfIlluminationMap(cubeFaces[3]);
220 materials[RIGHT] .setSelfIlluminationMap(cubeFaces[2]);
221 // Top/Bottom inversés : axe Y JavaFX opposé à la convention equi2cube
222 materials[TOP] .setSelfIlluminationMap(cubeFaces[5]); // equi Bottom → face Top
223 materials[BOTTOM].setSelfIlluminationMap(cubeFaces[4]); // equi Top → face Bottom
224
225 for (int i = 0; i < 6; i++) {
226 materials[i].setDiffuseMap(null);
227 materials[i].setSpecularMap(null);
228 materials[i].setDiffuseColor(Color.BLACK);
229 }

234 private Image resizeToEquirectangular(Image source, int targetWidth, int targetHeight) {
235 int sw = (int) source.getWidth();
236 int sh = (int) source.getHeight();
237 if (sw == targetWidth && sh == targetHeight) return source;

242 Logger.getLogger(PanoramicCube.class.getName()).log(Level.INFO,
243 "📐 Redimensionné : {0}×{1} → {2}×{3} (Bicubic)",
244 new Object[]{sw, sh, targetWidth, targetHeight});
245 return resized;
246 } catch (Exception e) {
247 Logger.getLogger(PanoramicCube.class.getName()).log(Level.WARNING,
248 "⚠️ GPU échoué, fallback CPU : {0}", e.getMessage());
249 return resizeToEquirectangularCPU(source, targetWidth, targetHeight);
250 }

253 private Image resizeToEquirectangularCPU(Image source, int tw, int th) {
254 int sw = (int) source.getWidth();
255 int sh = (int) source.getHeight();
256 WritableImage out = new WritableImage(tw, th);
257 PixelReader reader = source.getPixelReader();
258 PixelWriter writer = out.getPixelWriter();
259 double rx = (double) sw / tw;
260 double ry = (double) sh / th;
261 for (int y = 0; y < th; y++) {
262 for (int x = 0; x < tw; x++) {
263 int sx = Math.min((int) (x * rx), sw - 1);
264 int sy = Math.min((int) (y * ry), sh - 1);
265 writer.setColor(x, y, reader.getColor(sx, sy));
266 }
267 }
268 return out;
269 }
