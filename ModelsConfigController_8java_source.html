164 private void setupDragAndDrop(TableView<ModelEntryWrapper> table, ObservableList<ModelEntryWrapper> data) {
165 table.setRowFactory(tv -> {
166 TableRow<ModelEntryWrapper> row = new TableRow<>();
167
168 // Gestion du drag (début du glissement)
169 row.setOnDragDetected(event -> {
170 if (!row.isEmpty()) {
171 Integer index = row.getIndex();
172 Dragboard db = row.startDragAndDrop(TransferMode.MOVE);
173 db.setDragView(row.snapshot(null, null));
174 ClipboardContent cc = new ClipboardContent();
175 cc.putString(index.toString());
176 db.setContent(cc);
177 event.consume();
178 }
179 });
180
181 // Gestion du drop over (survol pendant le glissement)
182 row.setOnDragOver(event -> {
183 Dragboard db = event.getDragboard();
184 if (db.hasString()) {
185 if (row.getIndex() != Integer.parseInt(db.getString())) {
186 event.acceptTransferModes(TransferMode.COPY_OR_MOVE);
187 event.consume();
188 }
189 }
190 });
191
192 // Gestion du drop entered (entrée dans la zone)
193 row.setOnDragEntered(event -> {
194 if (event.getGestureSource() != row && event.getDragboard().hasString()) {
195 row.setOpacity(0.3);
196 }
197 });
198
199 // Gestion du drop exited (sortie de la zone)
200 row.setOnDragExited(event -> {
201 if (event.getGestureSource() != row && event.getDragboard().hasString()) {
202 row.setOpacity(1);
203 }
204 });
205
206 // Gestion du drop (dépôt final)
207 row.setOnDragDropped(event -> {
208 Dragboard db = event.getDragboard();
209 if (db.hasString()) {
210 int draggedIndex = Integer.parseInt(db.getString());
211 ModelEntryWrapper draggedItem = data.remove(draggedIndex);
212
213 int dropIndex;
214 if (row.isEmpty()) {
215 dropIndex = data.size();
216 } else {
217 dropIndex = row.getIndex();
218 }
219
220 data.add(dropIndex, draggedItem);
221
222 // Mettre à jour les priorités
223 updatePriorities(data);
224
225 event.setDropCompleted(true);
226 table.getSelectionModel().select(dropIndex);
227 event.consume();
228 }
229 });
230
231 // Gestion de la fin du drag
232 row.setOnDragDone(event -> {
233 event.consume();
234 });
235
236 return row;
237 });
238 }

687 private void openURL(String url) {
688 try {
689 String os = System.getProperty("os.name").toLowerCase();
690 System.out.println("[URL] Ouverture de: " + url);
691 System.out.println("[URL] OS détecté: " + os);
692
693 if (os.contains("linux")) {
694 // Linux : essayer plusieurs méthodes
695 boolean opened = false;
696
697 // Méthode 1 : Chercher un navigateur commun
698 String[] browsers = {"firefox", "google-chrome", "chromium", "brave-browser", "vivaldi", "opera"};
699 for (String browser : browsers) {
700 try {
701 Process checkProcess = new ProcessBuilder("which", browser).start();
702 checkProcess.waitFor();
703 if (checkProcess.exitValue() == 0) {
704 System.out.println("[URL] Navigateur trouvé: " + browser);
705 new ProcessBuilder(browser, url).start();
